# ─── Test 1: Basic Commands ───────────────────────────────────────────
# Expected: Lists files in current directory
ls

# Expected: Lists files in long format with hidden files
ls -la

# Expected: Prints current working directory
pwd

# Expected: Prints "Hello from MyShell"
echo "Hello from MyShell"

# Expected: Prints today's date
date

# Expected: Prints current username
whoami

# ─── Test 2: Built-in Commands ────────────────────────────────────────
# Expected: Shows help text with available commands and OS concepts
help

# Expected: Changes to /tmp, then pwd confirms it
cd /tmp
pwd

# Expected: Changes back to home directory
cd ~
pwd

# Expected: Changes back to project directory
cd - || cd /tmp

# Expected: Error message "No such file or directory" (should NOT crash)
cd /this/path/does/not/exist

# ─── Test 3: Environment Variables & Expansion ────────────────────────
# Expected: Set an environment variable
export COURSE="Operating Systems CSE311"
export SEMESTER="Fall 2026"

# Expected: Prints "Operating Systems CSE311 - Fall 2026"
echo "Enrolled in: $COURSE - $SEMESTER"

# Expected: Prints current user and home directory
echo "User: $USER Home: $HOME PID: $$"

# Expected: Displays all environment variables
env

# Expected: Unsets the variable
unset COURSE
echo "After unset: '$COURSE'"

# ─── Test 4: I/O Redirection (>, >>, <) ──────────────────────────────
# Expected: Redirects stdout to file (overwrite)
echo "Line 1: MyShell v2 I/O Redirection" > /tmp/myshell_test.txt
cat /tmp/myshell_test.txt

# Expected: Redirects stdout to file (append)
echo "Line 2: Appended output" >> /tmp/myshell_test.txt
cat /tmp/myshell_test.txt

# Expected: Redirects stdin from file
cat < /tmp/myshell_test.txt

# ─── Test 5: Pipe Commands (IPC) & Combined Redirection ──────────────
# Expected: Lists only .py files from current directory
ls -la | grep ".py"

# Expected: Counts the number of lines containing "root" in /etc/passwd
cat /etc/passwd | grep root | wc -l

# Expected: Pipeline with input and output redirection
cat < /tmp/myshell_test.txt | grep Line > /tmp/myshell_out.txt
cat /tmp/myshell_out.txt

# ─── Test 6: Logical Operators (&&, ||) & Exit Codes ─────────────────
# Expected: AND operator runs 2nd command only if 1st succeeds
true && echo "Condition 1 succeeded"

# Expected: 2nd command skipped because 1st fails
false && echo "This should NOT print"

# Expected: OR operator runs 2nd command only if 1st fails
false || echo "Condition 2 recovered via OR"

# Expected: 2nd command skipped because 1st succeeds
true || echo "This should NOT print"

# Expected: Exit code inspection via $?
ls /nonexistent_directory_12345
echo "Previous command exit code: $?"

ls *.py

# Expected: Expands single character wildcard
ls test_command?.txt
